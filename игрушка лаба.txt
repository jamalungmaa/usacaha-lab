<script>
const tg = window.Telegram.WebApp;
tg.expand();

// ВЫМЫШЛЕННЫЕ симптомы (вместе с "противоядиями")
const SYMPTOMS = [
  { id: 'itch',  n: '🟠 Зудящие пятна',      cures: ['mint'] },
  { id: 'glow',  n: '💡 Светящийся нос',     cures: ['dark', 'sleep'] },
  { id: 'hum',   n: '🎵 Гудение в ушах',     cures: ['lullaby'] },
  { id: 'hug',   n: '🤗 Внезапные обнимашки', cures: ['distance'] },
  { id: 'cryst', n: '💎 Хрустальные слёзы',  cures: ['mint'] },
  { id: 'crow',  n: '🐓 Кукареканье по утрам', cures: ['lullaby', 'sleep'] },
  { id: 'fear',  n: '🐙 Страх пельменей',    cures: ['therapy'] },
  { id: 'float', n: '🎈 Лёгкость в теле (буквально)', cures: ['soup'] },
  { id: 'echo',  n: '📢 Эхо-смех',           cures: ['lullaby', 'therapy'] },
];

const SPREAD = [
  { id: 'air',  n: '💨 Воздушные поцелуи' },
  { id: 'dish', n: '🍽 Общая тарелка' },
  { id: 'look', n: '👀 Вредоносный взгляд' },
  { id: 'wifi', n: '📶 Через Wi-Fi' },
];

// ВЫМЫШЛЕННЫЕ методы лечения. side = побочка, которая усиливает симптомы
const CURES = [
  { id: 'mint',     n: '🌿 Мятный компресс',      side: [] },
  { id: 'lullaby',  n: '🎶 Баюкальная колыбельная', side: [] },
  { id: 'sleep',    n: '😴 12 часов сна',          side: [] },
  { id: 'distance', n: '🧱 Карантин в домике',      side: ['hug'] },
  { id: 'soup',     n: '🍲 Суп от бабушки',        side: ['itch'] },
  { id: 'therapy',  n: '🛋 Разговор с психологом',  side: [] },
  { id: 'dark',     n: '🕶 Тёмные очки',            side: [] },
];

const PREFIX = ['Болезнь', 'Хворь', 'Ломота', 'Нытик', 'Зудосмет'];
const SUFFIX = ['грозная', 'полосатая', 'ночная', 'округлая', 'таинственная'];

let pickedSymptoms = [];
let pickedSpread = null;
let pickedCures = [];
let bestScore = +(localStorage.getItem('doctorScore') || 0);

function buildChips(containerId, items, type) {
  const box = document.getElementById(containerId);
  box.innerHTML = '';
  items.forEach(item => {
    const chip = document.createElement('div');
    chip.className = 'chip';
    chip.textContent = item.n;
    chip.onclick = () => {
      const arr = type === 'cure' ? pickedCures : pickedSymptoms;
      const max = type === 'cure' ? 3 : 3;
      const idx = arr.indexOf(item.id);
      if (idx >= 0) {
        arr.splice(idx, 1);
        chip.classList.remove('on');
      } else if (arr.length < max) {
        arr.push(item.id);
        chip.classList.add('on');
      } else {
        tg.showAlert('Можно выбрать не больше трёх!');
      }
      if (type === 'spread') {
        pickedSpread = item.id;
        document.querySelectorAll('#spreadChips .chip').forEach(c => c.classList.remove('on'));
        chip.classList.add('on');
      }
    };
    box.appendChild(chip);
  });
}

function makeDisease() {
  if (pickedSymptoms.length < 3) { tg.showAlert('Сначала выбери три симптома!'); return; }
  if (!pickedSpread) { tg.showAlert('Выбери способ передачи!'); return; }
  const name = PREFIX[Math.floor(Math.random() * PREFIX.length)] + ' ' +
               SUFFIX[Math.floor(Math.random() * SUFFIX.length)];
  const spread = SPREAD.find(s => s.id === pickedSpread).n;
  const symList = pickedSymptoms.map(id => SYMPTOMS.find(s => s.id === id).n).join(', ');
  document.getElementById('result').style.display = 'none';
  tg.showAlert('Придумана: «' + name + '»!\nСимптомы: ' + symList + '\nПередаётся: ' + spread + '\nТеперь подбери лечение и жми «Применить лечение!»');
}

function treat() {
  if (pickedSymptoms.length < 3 || !pickedSpread) { tg.showAlert('Сначала сгенерируй болезнь!'); return; }
  if (pickedCures.length === 0) { tg.showAlert('Выбери хотя бы один метод лечения!'); return; }

  let score = 0;
  const cured = [], useless = [], harmed = [];

  pickedCures.forEach(cid => {
    const cure = CURES.find(c => c.id === cid);
    const heals = pickedSymptoms.filter(sid => cure.cures.includes(sid));
    const hurts = cure.side.filter(sid => pickedSymptoms.includes(sid));
    if (heals.length) { score += heals.length * 2; cured.push(cure.n); }
    else useless.push(cure.n);
    if (hurts.length) { score -= hurts.length * 2; harmed.push(cure.n); }
  });

  const max = pickedSymptoms.length * 2;
  const pct = Math.max(0, Math.min(100, Math.round(score / max * 100)));
  let diag, emoji;
  if (pct >= 100)      { diag = 'Полное исцеление! Нобелевка доктора Чудака обеспечена.'; emoji = '🏆'; }
  else if (pct >= 60)  { diag = 'Пациент почти здоров. Почти.'; emoji = '😃'; }
  else if (pct >= 30)  { diag = 'Стало лучше... местами.'; emoji = '🤔'; }
  else if (pct > 0)    { diag = 'Пациент жив, но недоволен.'; emoji = '😬'; }
  else                 { diag = 'Хмуро. Симптомы обиделись и стали сильнее.'; emoji = '💥'; }

  let lines = [];
  if (cured.length)   lines.push('✅ Помогло: ' + cured.join(', '));
  if (useless.length) lines.push('⚪ Не помогло: ' + useless.join(', '));
  if (harmed.length)  lines.push('❌ Побочка усилила болезнь: ' + harmed.join(', '));

  if (pct > bestScore) {
    bestScore = pct;
    localStorage.setItem('doctorScore', bestScore);
  }

  const res = document.getElementById('result');
  res.style.display = 'block';
  res.innerHTML = '<div class="diag">' + emoji + ' ' + diag + '</div>' +
                  '<div id="score">' + pct + '%</div>' +
                  lines.join('<br>') +
                  '<br><small>Личный рекорд: ' + bestScore + '%</small>';
}

buildChips('symptomChips', SYMPTOMS, 'symptom');
buildChips('spreadChips', SPREAD, 'spread');
buildChips('cureChips', CURES, 'cure');
</script>
